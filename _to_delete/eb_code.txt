function parseFile(text,name){
  const t=text.trim();let rows=[],src='';
  if(t.startsWith('{')){const j=JSON.parse(t);const fs=j.features||[];src='USGS (GeoJSON)';
    for(const f of fs){const c=f.geometry?.coordinates,p=f.properties||{};if(!c||p.mag==null)continue;
      const tm=typeof p.time==='number'?p.time:Date.parse(p.time);
      rows.push([Math.round(tm/60000),+c[1],+c[0],+(c[2]??0),+p.mag,p.place||p.flynn_region||'',p.tsunami?1:0,p.alert?String(p.alert)[0]:'',f.id||'','usgs']);}}
  else{const lines=t.split(/\r?\n/).filter(l=>l.trim());const h=lines[0];
    if(h.startsWith('#')&&h.includes('|')){src='GEOFON / FDSN';const cols=h.slice(1).split('|').map(s=>s.trim().toLowerCase());
      const ix=n=>cols.findIndex(c=>c.startsWith(n));const iT=ix('time'),iLa=ix('latitude'),iLo=ix('longitude'),iD=ix('depth'),iM=ix('magnitude'),iP=ix('eventlocationname'),iId=ix('eventid');
      for(const l of lines.slice(1)){if(l.startsWith('#'))continue;const v=l.split('|');const m=parseFloat(v[iM]);if(isNaN(m))continue;
        let ts=v[iT];if(!/Z|[+-]\d\d:?\d\d$/.test(ts))ts+='Z';
        rows.push([Math.round(Date.parse(ts)/60000),+v[iLa],+v[iLo],+v[iD]||0,m,iP>=0?v[iP]:'',0,'',iId>=0?v[iId]:'',/geofon|gfz/i.test(h+name+l)?'geofon':'fdsn']);}}
    else if(/latitude/i.test(h)&&/mag/i.test(h)){src='USGS (CSV)';const cols=splitCSV(h).map(s=>s.toLowerCase());
      const ix=n=>cols.indexOf(n);const iT=ix('time'),iLa=ix('latitude'),iLo=ix('longitude'),iD=ix('depth'),iM=ix('mag'),iP=ix('place'),iId=ix('id');
      for(const l of lines.slice(1)){const v=splitCSV(l);const m=parseFloat(v[iM]);if(isNaN(m))continue;
        rows.push([Math.round(Date.parse(v[iT])/60000),+v[iLa],+v[iLo],+v[iD]||0,m,iP>=0?v[iP]:'',0,'',iId>=0?v[iId]:'','usgs']);}}
    else throw new Error('Das Format wurde nicht erkannt. Erwartet wird USGS-GeoJSON, USGS-CSV oder GEOFON im FDSN-Textformat.');}
  rows=rows.filter(r=>isFinite(r[0])&&isFinite(r[1])&&isFinite(r[2])&&isFinite(r[4]));
  if(!rows.length)throw new Error('In der Datei wurden keine Beben mit Magnitude gefunden.');
  const mm=Math.min(...rows.map(r=>r[4]));
  return {gen:Math.max(...rows.map(r=>r[0]))*60000,e:rows,src,label:name.replace(/\.[^.]+$/,'').slice(0,22),minmag:Math.max(0,Math.floor(mm*10)/10)};}
function splitCSV(l){const o=[];let cur='',q=false;for(let i=0;i<l.length;i++){const ch=l[i];
  if(q){if(ch==='"'){if(l[i+1]==='"'){cur+='"';i++;}else q=false;}else cur+=ch;}
  else if(ch==='"')q=true;else if(ch===','){o.push(cur);cur='';}else cur+=ch;}o.push(cur);return o;}
function loadFile(f){const msg=document.getElementById('impmsg');msg.className='';msg.textContent='Lese '+f.name+' …';
  const rd=new FileReader();rd.onload=()=>{try{const d=parseFile(String(rd.result),f.name);ST.custom=d;
      document.getElementById('minmag').max=Math.max(7.5,...d.e.map(r=>r[4]));
      setSet('custom');msg.textContent=`${d.e.length.toLocaleString('de-DE')} Beben aus ${d.src} geladen.`;}
    catch(e){msg.className='err';msg.textContent=e.message.startsWith('Das Format')||e.message.startsWith('In der')?e.message:'Die Datei ließ sich nicht lesen. Prüfe, ob sie vollständig gespeichert wurde.';}};
  rd.readAsText(f);}

function legend(){const s=document.getElementById('legmag');let x=6,o='';
  for(const m of [3,4,5,6,7]){const r=rMag(m);x+=r;o+=`<circle cx="${x}" cy="20" r="${r}" fill="#fff" stroke="#1B1B2F" stroke-width="1.2"/><text x="${x}" y="42" text-anchor="middle" font-size="11" fill="#1B1B2F" font-family="Lexend, sans-serif">${m}</text>`;x+=r+16;}
  s.innerHTML=o;}
function init(){
  build();buildMap();legend();
  document.getElementById('viewtoggle').addEventListener('click',ev=>{const b=ev.target.closest('button[data-view]');if(b)setView(b.dataset.view);});
  document.getElementById('sets').addEventListener('click',ev=>{const b=ev.target.closest('button[data-k]');if(b)setSet(b.dataset.k);});
  const mm=document.getElementById('minmag');mm.addEventListener('input',()=>{ST.min=+mm.value;updMin();rebuildQuakes();});
  for(const cb of document.querySelectorAll('input[data-layer]'))cb.addEventListener('change',()=>{ST.layers[cb.dataset.layer]=cb.checked;draw();});
  document.getElementById('prof').addEventListener('click',ev=>{const g=ev.target.closest('.col');if(g){const q=PROFQ[+g.dataset.i];if(q){stopSpin();select(q);}}});
  document.getElementById('prof').addEventListener('keydown',ev=>{if(ev.key!=='Enter'&&ev.key!==' ')return;const g=ev.target.closest('.col');if(g){ev.preventDefault();const q=PROFQ[+g.dataset.i];if(q){stopSpin();select(q);}}});
  document.getElementById('top').addEventListener('click',ev=>{const b=ev.target.closest('button[data-id]');if(!b)return;const q=ACTIVE.find(x=>String(x[8])===b.dataset.id);if(q){stopSpin();select(q);}});
  const sl=document.getElementById('tslider');sl.addEventListener('input',()=>{stopPlay();updTime();renderCount();renderTop();draw();});
  document.getElementById('play').addEventListener('click',()=>ST.play?stopPlay():startPlay());
  const fi=document.getElementById('file');fi.addEventListener('change',()=>{if(fi.files[0])loadFile(fi.files[0]);});
  const dr=document.getElementById('drop');
  dr.addEventListener('dragover',e=>{e.preventDefault();dr.classList.add('over');});
  dr.addEventListener('dragleave',()=>dr.classList.remove('over'));
  dr.addEventListener('drop',e=>{e.preventDefault();dr.classList.remove('over');const f=e.dataTransfer.files[0];if(f)loadFile(f);});
  setSet('month');
  // Start: stärkstes Beben des Monats auswählen
  const top=ACTIVE[ACTIVE.length-1];if(top){ST.sel=top;GL.rot=[-top[2],clamp(-top[1],-80,80)*0.6,0];renderInfo();renderTop();}
  let pv=null;try{pv=localStorage.getItem('erdbeben-view');}catch(e){}
  setView(pv==='globe'?'globe':'map');
  addEventListener('resize',()=>{if(GL.view==='globe'){globeSize();drawGlobe();}});
}
init();
